{{/* Hugo Image Processing Shortcode/Partial Usage as shortcode: {{ < img src="project-name/image.jpg" size="200x" quality="q75" alt="Alt text" class="custom-class" > }} Usage as partial: {{ partial "shortcodes/img.html" (dict "src" "image.jpg" "size" "200x" "quality" "q75" "alt" "Alt text" "class" "custom-class") }} Parameters: - src (required): Path to image within assets/img/ (without /img/ prefix) - size (optional): Resize dimensions like "200x", "800x", "400x600" - defaults to "400x" - quality (optional): Quality setting like "q75", "q85", "q100" - defaults to "q75" - alt (optional): Alt text for accessibility - defaults to filename - class (optional): CSS class for the img tag - defaults to empty - loading (optional): Loading attribute - defaults to "lazy" */}} {{/* Handle both shortcode (.Get) and partial (direct access) parameter access */}} {{- $src := "" -}} {{- $size := "" -}} {{- $quality := "" -}} {{- $alt := "" -}} {{- $class := "" -}} {{- $loading := "" -}} {{- if .Get -}} {{/* Called as shortcode */}} {{- $src = .Get "src" -}} {{- $size = .Get "size" | default "400x" -}} {{- $quality = .Get "quality" | default "q75" -}} {{- $alt = .Get "alt" | default (path.Base $src) -}} {{- $class = .Get "class" -}} {{- $loading = .Get "loading" | default "lazy" -}} {{- else -}} {{/* Called as partial */}} {{- $src = .src -}} {{- $size = .size | default "400x" -}} {{- $quality = .quality | default "q75" -}} {{- $alt = .alt | default (path.Base $src) -}} {{- $class = .class -}} {{- $loading = .loading | default "lazy" -}} {{- end -}} {{- if not $src -}} {{- errorf "img shortcode/partial requires 'src' parameter" -}} {{- end -}} {{- $imageResource := false -}} {{- $imageUrl := printf "/img/%s" $src -}} {{/* Try to get the image from assets first */}} {{- with resources.Get (printf "img/%s" $src) -}} {{- $imageResource = . -}} {{- end -}} {{- if $imageResource -}} {{- if eq $imageResource.ResourceType "image" -}} {{- $resizeParams := printf "%s %s" $size $quality -}} {{- $image := $imageResource.Resize $resizeParams -}} {{- else -}} {{/* Non-image resource (like SVG), use fallback */}} {{- end -}} {{- else -}} {{/* Fallback to static folder */}} {{- end -}}
